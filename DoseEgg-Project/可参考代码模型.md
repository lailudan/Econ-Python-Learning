
import pandas as pd

# 假设这是用户的本地数据
data = {
    'time': ['06:00', '06:45', '12:00', '12:50'],
    'event': ['Medication', 'Symptom_Good', 'Medication', 'Symptom_Good']
}
df = pd.DataFrame(data)
df['time'] = pd.to_datetime(df['time'])

# 计算差值 (Lead Time)
# 逻辑：找出每一对 Med 和随后的 Good 之间的时间差
df['lead_time'] = df['time'].diff().dt.total_seconds() / 60
kick_in_samples = df[df['event'] == 'Symptom_Good']['lead_time']

print(f"平均起效时间: {kick_in_samples.mean()} 分钟")
